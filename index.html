Skema Relasi Database (ERD) & Alur Keamanan

Struktur tabel Google Sheets & Mekanisme Pencegahan Data Leakage Juri

// TABEL: CONFIG
- key (PK): STRING
- value: TEXT (Base64 Logo/Kop / Setting)
// TABEL: USERS
- user_id (PK): STRING
- username: STRING
- role: ENUM('SUPER_ADMIN','PANITIA','JURI','PESERTA')
// TABEL: LOMBA
- lomba_id (PK): STRING
- nama_lomba: STRING
- jumlah_juri_target: NUMBER (Dinamis: 1, 2, 3, 5, 7...)
- show_ranking_mode: ENUM('ALWAYS', 'ALL_DONE')
- kriteria_json: JSONString [{id, nama, bobot}]
// TABEL: PESERTA
- peserta_id (PK): STRING
- no_pendaftaran: STRING
- nama_lengkap: STRING
- asal_sekolah: STRING
- lomba_id (FK): STRING
- status_verifikasi: STRING
// TABEL: JURI_PLOTTING
- plot_id (PK): STRING
- lomba_id (FK): STRING
- juri_id (FK): STRING
- peserta_id (FK): STRING (Opsional)
// TABEL: NILAI
- nilai_id (PK): STRING
- lomba_id (FK): STRING
- peserta_id (FK): STRING
- juri_id (FK): STRING
- skor_json: JSONString {"k1":85, "k2":90}
- total_skor: NUMBER
- status: ENUM('DRAFT', 'SUBMITTED')
